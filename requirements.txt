# ---------------------------------------------------------------------
# 1. SOFTWARE TO INSTALL ON YOUR MACHINE
# ---------------------------------------------------------------------
Node.js            >= 20  (tested on 22.x)      https://nodejs.org
npm                >= 10  (comes with Node)
Git                any recent                   https://git-scm.com
A SQL client       psql  OR  pgAdmin  OR  DBeaver  (to run the .sql files)

PostgreSQL         16   local install (or Neon/Supabase cloud). Needs the pgcrypto extension; pgvector only for the optional bonus
Neo4j              5.x  Neo4j Aura Free (cloud, easiest) or Neo4j Desktop (local)
# Docker is NOT required.

# ---------------------------------------------------------------------

# ---------------------------------------------------------------------
# 6. ENVIRONMENT VARIABLES  (copy .env.example to .env.local, never commit it)
# ---------------------------------------------------------------------
DATABASE_URL      postgres://postgres:YOUR_PASSWORD@localhost:5432/supplychain
NEO4J_URI         neo4j://localhost:7687
NEO4J_USER        neo4j
NEO4J_PASSWORD    your Neo4j password
NEO4J_DATABASE    (optional)
JWT_SECRET        long random string, 32+ chars   (e.g. openssl rand -hex 32)

# ---------------------------------------------------------------------
# 7. PORTS USED
# ---------------------------------------------------------------------
3000   Next.js app            http://localhost:3000
5432   PostgreSQL
7474   Neo4j browser          http://localhost:7474
7687   Neo4j Bolt (driver)

# ---------------------------------------------------------------------
# 8. INSTALL AND RUN (quick version, full steps are in README.md)
# ---------------------------------------------------------------------
npm install
cp .env.example .env.local        # Windows PowerShell: copy .env.example .env.local
# create the supplychain database in Postgres and a Neo4j instance (Aura or Desktop), fill .env.local
npm run neo4j-setup
# ...load the PostgreSQL files from db/postgres (see README section 6)...
npm run create-admin -- admin@example.com "StrongPassword123" "Admin"
npm run dev

# ---------------------------------------------------------------------
# 9. NPM SCRIPTS
# ---------------------------------------------------------------------
npm run dev            # development server
npm run build          # production build
npm run start          # run the production build
npm run typecheck      # TypeScript check
npm run create-admin   # create the first admin user
npm run neo4j-setup    # create Neo4j constraints and indexes
